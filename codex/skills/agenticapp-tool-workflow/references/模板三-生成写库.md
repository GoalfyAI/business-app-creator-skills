# 模板三：生成写库

读应用库里的档案，让模型生成一份内容，写进应用后端预建好的那一行。这是更新后端已建行的写法：后端建行并填身份列，路线只按入口带入的 id 更新 agent 列，改到的行数不是 1 就返回 `blocked`「未入库」（`agenticapp-tool-data-template`「谁建行、谁写列」）。

照抄前：

- 把 [读写应用库与反读自查.md](读写应用库与反读自查.md) 的 `db()` 复制进脚本；
- 事件契约照 [事件契约与文件产物.md](事件契约与文件产物.md) 抄，三条都要；
- 只换表名、列名、业务键、入口字段、FA 的 `mr_name` 和 `stage_name`。

```python
async def run(input, ctx):
    await emit_business_event(event_type="stage_started", event_key="draft_started",
                              payload={"stage_key": "draft", "stage_name": "正在生成草稿"})

    _, profiles, _, _ = await db("SELECT category, city FROM merchant_profile LIMIT 1", why="生成时要用商家档案做本地化")
    if not profiles:
        await emit_business_event(event_type="stage_failed", event_key="draft_failed",
                                  payload={"stage_key": "draft", "error_code": "profile_missing", "message": "商家档案未填写"})
        return {"status": "needs_input", "missing": ["merchant_profile"], "draft_title": "", "draft_body": ""}

    draft = await tool(
        "<生成 FA 的 mr_name>",
        query="把选题写成一篇正文草稿",
        topic=input["topic"],
        profile_json=json.dumps(profiles[0], ensure_ascii=False),
        _rationale="由模型生成正文",
        _output={"type": "object", "required": ["status", "note", "draft_title", "draft_body", "hashtags"],
                 "properties": {"status": {"type": "string", "enum": ["completed", "partial", "blocked"]},
                                "note": {"type": "string"}, "draft_title": {"type": "string"}, "draft_body": {"type": "string"},
                                "hashtags": {"type": "array", "items": {"type": "string"}}}},
    )
    if draft["status"] != "completed":
        await emit_business_event(event_type="stage_failed", event_key="draft_failed",
                                  payload={"stage_key": "draft", "error_code": "draft_incomplete", "message": draft["note"]})
        return {"status": draft["status"], "note": draft["note"], "draft_title": "", "draft_body": ""}

    if not ctx.dry_run:  # 冒泡跳过写库，真跑照常写
        ok, _, updated, err = await db(
            "UPDATE note_records SET title = :title, content = :content, status = 'draft', updated_by = 'agent', "
            "last_run_id = :run_id, updated_at = now(), row_version = row_version + 1 WHERE id = :id",
            mode="writer", params={"title": draft["draft_title"], "content": draft["draft_body"], "run_id": ctx.run_id, "id": input["note_id"]},
            why="把草稿写进后端预建的候选行，使用者在应用里决定发不发",
        )
        if not ok or updated != 1:
            reason = err or "草稿行不存在"
            await emit_business_event(event_type="stage_failed", event_key="draft_failed",
                                      payload={"stage_key": "draft", "error_code": "draft_not_saved", "message": f"未入库：{reason}"})
            return {"status": "blocked", "note": f"未入库：{reason}", "draft_title": "", "draft_body": ""}

    await emit_business_event(event_type="stage_result", event_key="draft_completed",
                              payload={"stage_key": "draft", "summary": draft["draft_title"]})
    return {"status": "completed", "draft_title": draft["draft_title"], "draft_body": draft["draft_body"], "hashtags": draft["hashtags"]}
```
